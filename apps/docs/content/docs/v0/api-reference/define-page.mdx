## Signature

```ts
function definePage<TPath extends string>(
	path: TPath,
): {
	// With loader
	<TTemplate extends keyof TemplateRegistry, TLoaderData>(config: {
		template: TTemplate;
		loader: (ctx: Context<InferParams<TPath>>) => TLoaderData | Promise<TLoaderData>;
		defineSlots: (args: { loaderData: Awaited<TLoaderData> }) => SlotMap;
		meta?: MetaInfo | ((args: { loaderData: Awaited<TLoaderData> }) => MetaInfo);
	}): PageModule;

	// Without loader
	<TTemplate extends keyof TemplateRegistry>(config: {
		template: TTemplate;
		defineSlots: () => SlotMap;
		meta?: MetaInfo;
	}): PageModule;
};
```

`definePage` is curried. You call it with the route path first, then with the page config.

## Returns

A page module object branded with `[RouteKind]: "page"`.

## Usage

### Without loader

```tsx
import { definePage } from "@sundayceo/framework";

export default definePage("/about")({
	template: "default",
	meta: { title: "About" },
	defineSlots: () => ({
		main: <p>About page content.</p>,
	}),
});
```

### With loader

```tsx
export default definePage("/blog/[slug]")({
	template: "default",
	loader: ({ params }) => ({ slug: params.slug }),
	meta: ({ loaderData }) => ({ title: loaderData.slug }),
	defineSlots: ({ loaderData }) => ({
		main: <h1>{loaderData.slug}</h1>,
	}),
});
```
