## Signature

```ts
function createApp<
	TPlatform = unknown,
	TCustom extends Record<string, unknown> = Record<string, unknown>,
>(config: AppConfig<TCustom, TPlatform>): AppConfig<TCustom, TPlatform>;
```

## Parameters

### `config`

## Returns

The same config object, typed as `AppConfig<TCustom, TPlatform>`.

## Usage

```ts title="src/app.ts"
import { createApp } from "@sundayceo/framework";

type Platform = { env: Record<string, unknown>; ctx: unknown };

export const app = createApp<Platform>({
	context: (request, platform) => ({
		appName: "my-app",
		db: platform.env.DB,
	}),
	onError: (error, request) => {
		console.error(`[${request.url}]`, error);
	},
});

// Register for type inference across the app
declare module "@sundayceo/framework" {
	interface Register {
		app: typeof app;
	}
}
```
